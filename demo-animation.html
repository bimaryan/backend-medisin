protected static function booted(): void {
    static::creating(function (Pasien $pasien): void {
        if (empty($pasien->no_rekam_medik)) {
            $pasien->no_rekam_medik = static::generateNextNoRekamMedik();
        }
    });
}

public static function generateNextNoRekamMedik(): string {
    $allRms = static::query()->pluck('no_rekam_medik')
        ->filter(fn ($rm) => is_string($rm) && preg_match('/^RM(\\d+)$/i', $rm))
        ->map(fn ($rm) => (int) preg_replace('/^RM/i', '', $rm));
    $maxNumber = $allRms->max() ?? 0;
    return sprintf('RM%04d', $maxNumber + 1);
}

public function setNikAttribute(?string $value): void {
    $this->attributes['nik'] = Crypt::encryptString($value);
    $this->attributes['nik_hash'] = hash('sha256', trim($value));
}

SecurityService: Enkripsi AES-256-CBC Payload

CRYPTOGRAPHY SERVICE 
// 1. Enkripsi: IV acak 16 byte per request → Output Base64(IV + Ciphertext)
public function encrypt(string $value): string {
    $iv = openssl_random_pseudo_bytes(16);
    $cipher = openssl_encrypt($value, 'AES-256-CBC', $this->key, OPENSSL_RAW_DATA, $iv);
    return base64_encode($iv . $cipher);
}

// 2. Dekripsi multi-format (Web Crypto frontend, Postman CryptoJS, Laravel Crypt)
public function decrypt(?string $val): ?string {
    if (!$val) return $val;
    try { return Crypt::decryptString($val); } catch (\Throwable) {}
    if (str_starts_with($val, 'U2FsdGVkX1')) return $this->decryptCryptoJS($val, $this->key);
    $raw = base64_decode($val, true);
    if ($raw && strlen($raw) > 16) {
        return openssl_decrypt(substr($raw, 16), 'AES-256-CBC', $this->key, OPENSSL_RAW_DATA, substr($raw, 0, 16));
    }
    return $val; // Fallback plaintext
}

Controller: Agregasi Jml Kunjungan & Pencarian No RM

app/Http/Controllers/Api/PasienController.php & KunjunganController.php
CONTROLLER LAYER 
// Catatan #1: Jumlah Kunjungan dihitung dari berapa kali kunjungan dibuat (Anti N+1)
public function index(Request $request): JsonResponse {
    $query = Pasien::query()->withCount('kunjungans as jml_kunjungan');

    if ($search = $request->input('search')) {
        $hash = hash('sha256', trim((string) $search));
        $query->where(fn ($q) => $q->where('nama_pasien', 'like', "%{$search}%")
            ->orWhere('no_rekam_medik', 'like', "%{$search}%")
            ->orWhere('nik_hash', $hash));
    }
    return response()->json(['success' => true, 'data' => PasienResource::collection($query->get())]);
}

// Catatan #3: Endpoint khusus modal "Buat Kunjungan" (Input No RM → Muncul Data Pasien)
public function findByRekamMedik(string $no_rekam_medik): JsonResponse {
    $pasien = Pasien::where('no_rekam_medik', $no_rekam_medik)
        ->withCount('kunjungans as jml_kunjungan')->firstOrFail();
    return response()->json(['success' => true, 'data' => new PasienResource($pasien)]);
}
